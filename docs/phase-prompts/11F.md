```python
import subprocess
from pathlib import Path

BASE = "c6a3aeb"


def show(path):
    return subprocess.run(["git", "show", f"{BASE}:{path}"], capture_output=True, check=True).stdout.decode("utf-8").replace("\r\n", "\n")


def between(text, start, stop):
    i = text.index(start)
    j = text.index(stop, i + len(start))
    return text[i:j].rstrip("\n") + "\n"


def write(path, text):
    Path(path).write_text(text, encoding="utf-8", newline="\n")


status = show("docs/STATUS.md")
body = status.split("\n---\n", 1)[1].lstrip("\n").replace(" do candidato ", " do autor ")
write("docs/status-history.md",
      "# Registro por fase\n\n"
      "Registro completo de cada fase, movido sem alteração de `docs/STATUS.md` em 2026-11-09 (a única troca foi a "
      "palavra usada para o dono do repositório, que passou a ser \"autor\"). Cada seção traz a saída bruta do verificador da fase, os testes criados ou alterados, "
      "o que não foi feito e as perguntas ao SPEC. As seções estão da mais recente para a mais antiga.\n\n---\n\n" + body)

readme = show("README.md")
parts = [
    ("Resultados da comparação de modelos (v1)",
     between(readme, "### Resultados da avaliação: comparação de modelos (3 execuções por modelo, 2026-11-08)",
             "### Resultados da avaliação (execução `full-1`, 2026-11-07)")),
    ("Resultados da execução full-1 (v1)",
     between(readme, "### Resultados da avaliação (execução `full-1`, 2026-11-07)", "#### 1. Injeção Segura da Chave de API")),
    ("Matriz dos cenários oficiais medida na full-1",
     between(readme, "## 6. Matriz de Cobertura dos Cenários Oficiais", "## 7. Glossário de Defesa Técnica")),
    ("Troca da base escrita à mão pelo corpus oficial",
     between(readme, "* **Por que trocamos a base**:", "* **Pipeline Oficial de RAG**:")),
    ("Perguntas e respostas que estavam no README",
     between(readme, "## 7. Glossário de Defesa Técnica", "## 8. Estrutura do Repositório")),
]
out = ["# Histórico de resultados\n\n"
       "Trechos retirados do README em 2026-11-09, quando ele foi reorganizado para mostrar só os resultados vigentes. "
       "O texto de cada trecho está como estava no README do commit `c6a3aeb`; os números continuam valendo para as "
       "execuções a que se referem (registros em `reports/eval_runs/`). Os resultados vigentes estão no README.\n"]
for title, text in parts:
    out.append(f"\n---\n\n## {title}\n\n{text.rstrip(chr(10)).rstrip('-').rstrip(chr(10))}\n")
write("docs/history.md", "".join(out))

write("docs/STATUS.md",
      "# STATUS DO REPOSITÓRIO\n\n"
      "**Data de geração**: 09/11/2026 | **Branch**: `main`\n\n"
      "## Resumo\n\n"
      "- O sistema foi construído em fases. Cada fase tem um prompt em `docs/phase-prompts/`, um verificador em "
      "`scripts/verify_phase*.py` (roda sem chave de API e reroda os verificadores anteriores) e um commit próprio; o CI "
      "roda ruff, pytest, o build da imagem e um smoke test em `/health` a cada push.\n"
      "- Fases 1 a 8: supervisor com function calling, RAG sobre páginas oficiais, autenticação, memória, transbordo humano, "
      "avaliação com juiz, observabilidade e Docker.\n"
      "- Fase 9: correções da auditoria simulada (cartão e CPF fora do disco, chamada de ferramenta órfã, mock sem conta "
      "inventada, imagem em produção, logs sem texto da conversa).\n"
      "- Fase 10: clima e câmbio por API, guardrail híbrido (padrões e classificador por LLM), conjunto de avaliação v2 e "
      "execuções com modelo real.\n"
      "- Fase 11: resposta inválida do classificador passa a bloquear; holdout do guardrail commitado antes da única "
      "execução (`author-1`); o runner registra a camada do guardrail por turno; tier e modelos nos relatórios; "
      "documentação reorganizada.\n"
      "- Resultados vigentes: README e `reports/`. Resultados antigos: `docs/history.md`.\n"
      "- Registro completo de cada fase (saída bruta dos verificadores, testes, perguntas ao SPEC): `docs/status-history.md`.\n\n"
      "---\n")
```
